Setup Steps (Google Apps Script)

  1. Create a new Google Sheet and rename the tab at the bottom to Expenses.
  2. In Row 1, paste these 11 headers: ID | Date | Item | Category | PaymentMethod | Amount | SubmittedBy | Department | Status | InvoiceName | InvoiceUrl | Notes
  3. Open Extensions → Apps Script, paste the script below (which also auto-saves uploaded JPG/PDF invoices to a Google Drive folder named SpendWise_Invoices), and save.
  4. Click Deploy → New deployment → Web app (Execute as: Me, Access: Anyone).
  5. Log in as Admin → open Google Sheets & Security tab → paste your Web App URL and click Test Connection & Sync.
Full CRUD + Drive Invoice Google Apps Script 
const SHEET_NAME = "Expenses";
const DRIVE_FOLDER_NAME = "SpendWise_Invoices";

function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) return jsonResponse({ status: "error", message: "Sheet 'Expenses' not found" });
  
  const data = sheet.getDataRange().getValues();
  const rows = [];
  for (let i = 1; i < data.length; i++) {
    if (!data[i][1] && !data[i][2]) continue;
    rows.push({
      id: String(data[i][0] || i),
      date: formatDate(data[i][1]),
      item: String(data[i][2]),
      category: String(data[i][3]),
      paymentMethod: String(data[i][4]),
      amount: Number(data[i][5]) || 0,
      submittedBy: String(data[i][6] || "Admin"),
      department: String(data[i][7] || "General"),
      status: String(data[i][8] || "Approved"),
      invoiceName: String(data[i][9] || ""),
      invoiceData: String(data[i][10] || ""),
      notes: String(data[i][11] || "")
    });
  }
  return jsonResponse({ status: "success", data: rows });
}

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const payload = JSON.parse(e.postData.contents);
  const action = payload.action || "add";
  
  if (action === "delete") {
    const idsToDelete = new Set((payload.ids || [payload.id]).map(String));
    const data = sheet.getDataRange().getValues();
    for (let i = data.length - 1; i >= 1; i--) {
      if (idsToDelete.has(String(data[i][0]))) {
        sheet.deleteRow(i + 1);
      }
    }
    return jsonResponse({ status: "success", action: "deleted" });
  }
  
  if (action === "update") {
    const data = sheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(payload.id)) {
        const driveUrl = saveInvoiceToDrive(payload);
        sheet.getRange(i + 1, 2, 1, 11).setValues([[
          payload.date,
          payload.item,
          payload.category,
          payload.paymentMethod,
          Number(payload.amount),
          payload.submittedBy || "Admin",
          payload.department || "General",
          payload.status || "Approved",
          payload.invoiceName || "",
          driveUrl || String(data[i][10] || ""),
          payload.notes || ""
        ]]);
        break;
      }
    }
    return jsonResponse({ status: "success", action: "updated" });
  }

  if (action === "bulkSync") {
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, 12).clearContent();
    }
    const rows = (payload.entries || []).map(item => [
      item.id, item.date, item.item, item.category,
      item.paymentMethod, Number(item.amount),
      item.submittedBy || "Admin", item.department || "General",
      item.status || "Approved", item.invoiceName || "",
      (item.invoiceData && item.invoiceData.startsWith("http")) ? item.invoiceData : "",
      item.notes || ""
    ]);
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, 12).setValues(rows);
    }
    return jsonResponse({ status: "success", count: rows.length });
  }
  
  const driveUrl = saveInvoiceToDrive(payload);
  sheet.appendRow([
    payload.id,
    payload.date,
    payload.item,
    payload.category,
    payload.paymentMethod,
    Number(payload.amount),
    payload.submittedBy || "Admin",
    payload.department || "General",
    payload.status || "Approved",
    payload.invoiceName || "",
    driveUrl || "",
    payload.notes || ""
  ]);
  return jsonResponse({ status: "success", entry: payload, invoiceUrl: driveUrl });
}

function saveInvoiceToDrive(payload) {
  if (!payload.invoiceData || !payload.invoiceData.startsWith("data:")) return "";
  try {
    const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
    const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);
    const parts = payload.invoiceData.split(",");
    const mimeMatch = parts[0].match(/:(.*?);/);
    const mimeType = mimeMatch ? mimeMatch[1] : "application/octet-stream";
    const bytes = Utilities.base64Decode(parts[1]);
    const blob = Utilities.newBlob(bytes, mimeType, (payload.id || "INV") + "_" + (payload.invoiceName || "receipt"));
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return file.getUrl();
  } catch (err) {
    return "";
  }
}

function formatDate(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  return String(val).slice(0, 10);
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
